test("third-order influence audits are canonical, non-compensatory, and cross-domain", () => {
  const audits = [...new Set(facts.filter(fact => fact.p === "isa" && fact.o === "ThirdOrderInfluenceAudit").map(fact => fact.s))];
  assert.equal(audits.length, 6);
  const dimensions = ["standing", "boundary", "agency", "telos", "correction"];
  const domains = new Set();
  const verdicts = [];
  for (const subject of audits) {
    const record = Object.fromEntries(facts.filter(fact => fact.s === subject).map(fact => [fact.p, fact.o]));
    domains.add(record.domain);
    verdicts.push(record.verdict);
    for (const dimension of dimensions) {
      assert.ok(record[dimension], `${subject} ${dimension}`);
      assert.match(record[`${dimension}Result`], /^(pass|watch|fail)$/u, `${subject} ${dimension} result`);
    }
    const results = dimensions.map(dimension => record[`${dimension}Result`]);
    if (results.includes("fail")) assert.equal(record.verdict, "illegitimate", subject);
    else if (results.includes("watch")) assert.equal(record.verdict, "conditional", subject);
    else assert.equal(record.verdict, "legitimate", subject);
  }
  assert.equal(domains.size, 6);
  assert.deepEqual(verdicts.sort(), ["conditional", "conditional", "conditional", "illegitimate", "legitimate", "legitimate"]);
  assert.equal(hasFact("audit.claimipedia.thirdOrder.recommender", "boundaryResult", "fail"), true);
  assert.equal(hasFact("audit.claimipedia.thirdOrder.selfAuthoredLearning", "agencyResult", "pass"), true);
  assert.equal(hasFact("audit.claimipedia.thirdOrder.games", "verdict", "conditional"), true);
});

test("real server serves every custom claim route through the versioned SPA shell", async t => {
  const dataDir = mkdtempSync(join(tmpdir(), "claimipedia-king-way-kingdom-"));
  const started = await startClaimipediaServer({ dataDir, port: 0 });
  t.after(async () => {
    await new Promise(resolve => started.server.close(resolve));
    await started.service.drainEvaluationJobs();
    rmSync(dataDir, { recursive: true, force: true });
  });
  const stateResponse = await fetch(new URL("api/state", started.url));
  assert.equal(stateResponse.status, 200);
  const canonicalState = await stateResponse.json();
  const kingdomVersion = canonicalState.claimVersions.find(item => item.subject === "claimVersion.claimipedia.loadBearing.kingdom.v1");
  assert.equal(kingdomVersion.workedCaseTitle, "When a trusted church leader is accused of abusing a child");
  assert.match(kingdomVersion.workedCaseCorrection, /independent review|restitution/u);
  assert.equal(canonicalState.thirdOrderAudits.length, 6);
  assert.equal(canonicalState.thirdOrderAudits.find(item => item.subject === "audit.claimipedia.thirdOrder.games").verdict, "conditional");
  assert.ok(canonicalState.claims.some(item => item.subject === "claim.claimipedia.games.restorativePotential"));
  assert.equal(canonicalState.thirdOrderAudits.find(item => item.subject === "audit.claimipedia.thirdOrder.recommender").verdict, "illegitimate");
  const revisionBeforeDraft = started.service.revision;
  const draft = await started.service.proposeThirdOrderAudit({
    actor: "actor.claimipedia.author",
    expectedRevision: revisionBeforeDraft,
    domain: "Housing allocation",
    title: "A transparent housing triage process",
    intervention: "Applicants and public-interest reviewers inspect and challenge a rules-based priority order.",
    affectedParties: "Applicants, dependants, housing staff, people unable to complete the process, and residents affected by allocation.",
    standing: "Support is not conditioned on moral worth or agreement.", standingResult: "pass", standingEvidence: "Published eligibility rules and applicant testimony.",
    boundary: "Anti-discrimination and emergency safeguards constrain allocation.", boundaryResult: "pass", boundaryEvidence: "Policy controls and audit samples.",
    agency: "Applicants can inspect, correct, appeal, and use assisted routes.", agencyResult: "pass", agencyEvidence: "Appeal records and accessibility testing.",
    telos: "The process serves safe housing, though scarcity can distort priorities.", telosResult: "watch", telosEvidence: "Outcome data are incomplete for people who leave the queue.",
    correction: "Independent appeal can change rank and remedy an allocation.", correctionResult: "pass", correctionEvidence: "Appeal decisions and restitution records.",
    dissent: "Some affected people argue that any queue normalises an unjust shortage.",
    uncertainty: "Long-term outcomes for unsuccessful and non-digital applicants are not yet known.",
    rationale: "The case tests a public allocation field without treating procedural transparency as sufficient.",
  });
  assert.equal(draft.suggestedVerdict, "conditional");
  assert.equal(started.service.revision, revisionBeforeDraft, "proposal must not change the canonical revision");
  const stateWithDraft = started.service.state();
  assert.equal(stateWithDraft.thirdOrderAuditDrafts.length, 1);
  assert.equal(stateWithDraft.thirdOrderAuditDrafts[0].dissent, "Some affected people argue that any queue normalises an unjust shortage.");
  assert.equal(stateWithDraft.thirdOrderAudits.length, 6, "canonical audits remain unchanged");
  for (const route of [
    "/claim/jesus-is-king",
    "/claim/follow-me-third-order-influence",
    "/claim/the-kingdom",
    "/claim/claim.biblicalResurrection.simulationRupture",
    "/application/kingdom",
    "/application/kingdom-directory",
    "/application/kingdom-rooms",
    "/application/kingdom-safeguarding",
  ]) {
    assert.match(routesSource, new RegExp(`\\("[^"]+", "path", "${route.replaceAll(".", "\\.")}"`, "u"));
    const response = await fetch(new URL(route.slice(1), started.url));
    assert.equal(response.status, 200, route);
    assert.match(response.headers.get("content-type"), /text\/html/u, route);
    const html = await response.text();
    assert.match(html, /id="content"/u, route);
    assert.match(html, /src="\/app\.js\?v=[^"]+"/u, route);
  }
});
