const fs=require('fs'),path=require('path');const reports=[];for(const file of ['questions.html','the-field/index.html']){const html=fs.readFileSync('dist/'+file,'utf8'),base=new URL(file,'https://aaronalderman.com/');let checked=0;for(const m of html.matchAll(/\b(?:href|src)="([^"]+)"/g)){const url=new URL(m[1].replaceAll('&amp;','&'),base);if(url.origin!==base.origin)continue;const route=decodeURIComponent(url.pathname).slice(1),target=(!route||route.endsWith('/'))?route+'index.html':route;if(!fs.existsSync('dist/'+target))throw Error(file+': '+target);if(url.hash&&!fs.readFileSync('dist/'+target,'utf8').includes('id="'+decodeURIComponent(url.hash.slice(1))+'"'))throw Error(file+': missing '+url.hash);checked++;}reports.push({file,localResourcesAndFragments:checked});}console.log(JSON.stringify(reports));fs.writeFileSync('.build/questions-link-audit.json',JSON.stringify(reports,null,2));
