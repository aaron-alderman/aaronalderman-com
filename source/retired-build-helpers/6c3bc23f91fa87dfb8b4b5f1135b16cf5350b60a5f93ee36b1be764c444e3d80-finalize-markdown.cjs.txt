const fs=require('fs'),crypto=require('crypto'),p=JSON.parse(fs.readFileSync('.build/markdown-polish.json','utf8'));
const style=p.changes.find(c=>c.file==='source/hierarchy.rel');
const rules=[['.inspection-document-dual #hierarchy-preview,.inspection-document-dual #hierarchy-source',{'height':'min(75vh,900px)','overflow':'auto','overscroll-behavior':'contain','scrollbar-gutter':'stable','border-radius':'12px'}],['.inspection-document-dual .md-note',{'min-height':'42px'}],['.inspection-document-dual #hierarchy-preview:focus-visible,.inspection-document #hierarchy-source:focus-visible',{'outline':'2px solid #9fd6c3','outline-offset':'4px'}],['.inspection-document-dual #hierarchy-preview,.inspection-document-dual #hierarchy-source',{'height':'auto','max-height':'75vh'},'(max-width: 900px)'],['.markdown-preview a',{'color':'#206050'}]];
rules.forEach(([selector,decl,context],i)=>{const id='document.polish.'+i,rows=[[id,'isa','StyleRule'],[id,'projection','main'],[id,'selector',selector]];if(context)rows.push([id,'context',context]);Object.entries(decl).forEach(([key,value])=>rows.push([id,'declaration',id+'.'+key],[id+'.'+key,'property',key],[id+'.'+key,'value',value]));style.text+='\n'+rows.map(r=>'('+r.map(JSON.stringify).join(', ')+')').join('\n')+'\n';});
p.changes=p.changes.filter(c=>c.text!==fs.readFileSync(c.file,'utf8'));p.changes.forEach(c=>c.beforeSha256=crypto.createHash('sha256').update(fs.readFileSync(c.file)).digest('hex'));
fs.writeFileSync('.build/markdown-final.json',JSON.stringify(p));
