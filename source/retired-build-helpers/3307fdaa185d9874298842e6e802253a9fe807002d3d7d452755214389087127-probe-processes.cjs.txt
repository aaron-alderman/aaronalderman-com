const fs=require('fs'),assert=require('assert/strict'),{createKernel}=require('../bootstrap/temple-seed.cjs'),{lower}=require('../bootstrap/ports/algebra-lower.cjs'),{invoke}=require('../bootstrap/ports/programme.cjs'),{valuePorts}=require('../bootstrap/ports/value.cjs'),{computationalPorts}=require('../bootstrap/ports/browser-effects.cjs'),{createProcessCompiler}=require('../bootstrap/ports/process-source.cjs'),crypto=require('crypto');
const read=f=>JSON.parse(fs.readFileSync(f,'utf8')),sha=b=>crypto.createHash('sha256').update(b).digest('hex');
(async()=>{const receipt=read('dist/build-receipt.json'),owner=receipt.processLanguage[0];assert.equal(sha(fs.readFileSync(owner.file)),owner.sourceHash);assert.equal(sha(fs.readFileSync('dist/'+owner.output)),owner.moduleHash);const formatter=await lower({compiler:read('algebra/lowering.process.json'),definitions:read('algebra/operators.expression.json'),primitives:createKernel.operators,input:{entry:'sp.programme',queries:read('dist/'+owner.output).queries},mode:'model',ports:valuePorts(),limit:20000000}),compiler=await createProcessCompiler({read:async f=>fs.readFileSync(f,'utf8'),digest:sha},'bootstrap/process-frontend.json'),presentation=read('source/encounter/presentation.json'),rows=read('.build/process-inventory.json'),results=[];
