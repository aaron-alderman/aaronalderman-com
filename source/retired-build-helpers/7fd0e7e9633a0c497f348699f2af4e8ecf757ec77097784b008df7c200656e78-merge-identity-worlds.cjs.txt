const fs=require('fs'),cp=require('child_process');
const git=(...args)=>cp.execFileSync('git',['-c','safe.directory='+process.cwd().replaceAll('\\','/'),...args],{encoding:'utf8'});
const show=(rev,path)=>git('show',rev+':'+path);
const ours=JSON.parse(show('HEAD','algebra/publication.process.json')),theirs=JSON.parse(show('FETCH_HEAD','algebra/publication.process.json'));
const walk=(v,f)=>{if(v&&typeof v==='object'){f(v);Object.values(v).forEach(x=>walk(x,f));}};
let preload;walk(theirs,n=>{if(n.tag==='div'&&n.attrs?.hidden&&JSON.stringify(n).includes('IdentityTheme'))preload=n;});
walk(ours,n=>{if(n.tag==='img'&&n.attrs?.class==='hero-image')n.attrs.id='identity-background';if(n.tag==='script'&&n.attrs?.src?.startsWith?.('identity-runtime.js'))n.attrs.src='identity-runtime.js?v=identity-questions-worlds-3';if(n.tag==='body'&&JSON.stringify(n).includes('identity-switch'))n.children.push(preload);});
const effect=theirs.queries['identity.change'].find(x=>x.args?.ref==='#identity-background');ours.queries['identity.change'].splice(7,0,effect);
ours.queries['identity.runtime.build'][0].args.data.themes[3].image=['one','$t','image'];
fs.writeFileSync('algebra/publication.process.json',JSON.stringify(ours,null,2)+'\n');
let rel=show('HEAD','source/hierarchy.rel'),up=show('FETCH_HEAD','source/hierarchy.rel');
rel=rel.replace('"sepia(.8) hue-rotate(335deg)"','"none"').replace('"sepia(.7) saturate(1.2)"','"none"').replace('"sepia(.45) hue-rotate(245deg)"','"none"').replace('("identity.style.image.opacity", "value", ".48")','("identity.style.image.opacity", "value", ".78")').replace('("identity.style.image.mix-blend-mode", "value", "luminosity")','("identity.style.image.mix-blend-mode", "value", "normal")');
rel+='\n'+up.slice(up.indexOf('("identity.sourcerer", "image",'));
rel=rel.replaceAll('identity.father','identity.dad').replaceAll('identity.style.father','identity.style.dad').replaceAll('\\"father\\"','\\"dad\\"');fs.writeFileSync('source/hierarchy.rel',rel);
fs.writeFileSync('source/expedition.rel',show('FETCH_HEAD','source/expedition.rel').replace(/"stylesheetVersion", "[^"]+"/,'"stylesheetVersion", "identity-questions-worlds-3"'));
const h=JSON.parse(fs.readFileSync('algebra/hierarchy.process.json','utf8'));
for(const name of ['hierarchy.questions.build','hierarchy.identityQuestions.build'])walk(h.queries[name],n=>{if(n.tag==='h1'&&n.children?.[2]?.tag==='em'){n.children[1]=typeof n.children[1]==='string'?n.children[1]+' ':['concat',n.children[1],' '];}});
fs.writeFileSync('algebra/hierarchy.process.json',JSON.stringify(h,null,2)+'\n');
fs.writeFileSync('hierarchy.test.mjs',fs.readFileSync('hierarchy.test.mjs','utf8').replace("['key','label','copy','questionHref','questionCopy']","['key','label','copy','image','questionHref','questionCopy']"));
fs.writeFileSync('atlas-journey.test.mjs',fs.readFileSync('atlas-journey.test.mjs','utf8').replace("'father'","'dad'").replace('v=identity-worlds-1','v=identity-questions-worlds-3'));
for(const file of git('diff','--name-only','--diff-filter=U').trim().split(/\r?\n/).filter(x=>x.startsWith('dist/')))git('checkout','--ours','--',file);
console.log('Merged identity backgrounds, Dad naming, question routes and heading spacing.');
