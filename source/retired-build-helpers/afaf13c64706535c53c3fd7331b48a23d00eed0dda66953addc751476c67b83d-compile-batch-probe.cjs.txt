const fs=require('fs'),crypto=require('crypto'),{createProcessCompiler}=require('../bootstrap/ports/process-source.cjs'),read=f=>JSON.parse(fs.readFileSync(f,'utf8')),sha=b=>crypto.createHash('sha256').update(b).digest('hex');
(async()=>{fs.mkdirSync('.build/batch-02-compiled',{recursive:true});const results=[];for(const e of read('source/process-authoring/batch-02/ledger.json').entries){const c=await createProcessCompiler({read:async f=>fs.readFileSync(f,'utf8'),digest:sha},e.frontend),r=await c.compile(e.candidate,fs.readFileSync(e.candidate,'utf8'));require('assert/strict').equal(JSON.stringify(r.surface),JSON.stringify(read(e.oracle)));const text=JSON.stringify(r.surface,null,2)+'\n';fs.writeFileSync('.build/batch-02-compiled/'+e.name+'.json',text);results.push({name:e.name,moduleSha256:sha(text),sourceHash:r.sourceHash,steps:r.steps,origins:r.origins.length});}fs.writeFileSync('.build/batch-02-probe.json',JSON.stringify(results,null,2));console.log(results);})().catch(e=>{console.error(e);process.exitCode=1});
