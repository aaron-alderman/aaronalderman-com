import assert from 'node:assert/strict';
import crypto from 'node:crypto';
const base='https://aaronalderman.com/',paths=['assurance/reading-experience/','assurance/plain-language/','assurance/reading-experience.json','assurance/plain-language.json','assurance/topics.json','writing/make-the-magic-answerable/','expedition.css?v=reading-contrast-1','tooling/assurance/reading-experience-capture.json'];
const bodies=await Promise.all(paths.map(async path=>{const r=await fetch(base+path,{cache:'no-store'});assert.equal(r.status,200,path);return r.text();}));
assert.ok(bodies[0].includes('Can you read it, enlarge it, and find your way?'));assert.ok(bodies[1].includes('Can someone else understand what I mean?'));
const browser=JSON.parse(bodies[2]),language=JSON.parse(bodies[3]),topics=JSON.parse(bodies[4]),capture=JSON.parse(bodies[7]);
assert.equal(browser.status,'browser-observed');assert.equal(browser.counts.routes,20);assert.equal(browser.counts.missing,0);assert.equal(browser.counts.failed,2);assert.equal(browser.evidence.provenance.run,'42394118012');assert.equal(capture.snapshots.length,80);assert.equal(capture.journeys.length,20);
assert.equal(language.counts.routes,310);assert.equal(language.counts.candidates,73);assert.equal(language.counts.reviewed,0);
for(const id of ['reading-experience','plain-language'])assert.ok(topics.topics.some(t=>t.id===id));
assert.ok(bodies[5].includes('expedition.css?v=reading-contrast-1'));
assert.ok(bodies[6].includes('body:has(:is(.faith-threads,.reading-room,.faith-essay,.reading-essay)) .masthead{background:var(--reading-bg);color:var(--reading-text);'));
assert.equal(crypto.createHash('sha256').update(bodies[6]).digest('hex'),browser.evidence.artifactBindings['expedition.css']);
console.log(JSON.stringify({verified:paths,browser:browser.counts,language:language.counts,rawSnapshots:capture.snapshots.length,keyboardJourneys:capture.journeys.length}));
