import assert from 'node:assert/strict';
const base='https://aaronalderman.com/';
const paths=['temple/','claimipedia/','claimipedia/claimipedia.css?v=motif-3','assurance/reading-experience.json','assurance/reading-experience/'];
const responses=await Promise.all(paths.map(async path=>{const r=await fetch(base+path,{cache:'no-cache'});assert.equal(r.status,200,path);return await r.text();}));
assert.match(responses[0],/id="cmd-title"[^>]*style="overflow-wrap:break-word"/);
assert.match(responses[0],/class="nowrap"[^>]*style="white-space:normal"/);
assert.ok(responses[1].includes('claimipedia.css?v=motif-3'));
assert.match(responses[2],/\.claim-hero h1\s*\{[^}]*overflow-wrap:\s*break-word/s);
const r=JSON.parse(responses[3]);assert.equal(r.status,'browser-observed');assert.equal(r.counts.failed,0);assert.equal(r.counts.missing,0);assert.equal(r.evidence.provenance.run,'42398024098');
assert.equal(r.regressionChecks.length,2);assert.ok(r.regressionChecks.every(c=>c.status==='passed'));
console.log(JSON.stringify({live:true,run:r.evidence.provenance.run,counts:r.counts,regressionChecks:r.regressionChecks,urls:paths.map(p=>base+p)},null,2));
