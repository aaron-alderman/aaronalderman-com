import fs from 'node:fs';
import assert from 'node:assert/strict';
const before=JSON.parse(fs.readFileSync('.build/browser-evidence-42394118012/observations.json'));
const after=JSON.parse(fs.readFileSync('.build/browser-evidence-42398024098/observations.json'));
for(const profile of ['desktop','zoom-200-equivalent','zoom-400-equivalent']){
 const a=before.snapshots.find(s=>s.route==='temple/index.html'&&s.profile===profile),b=after.snapshots.find(s=>s.route===a.route&&s.profile===profile);
 const blocks=a.blocks.filter(n=>n.path.startsWith('[id="judgment'));
 for(const old of blocks){const next=b.blocks.find(n=>n.path===old.path);assert.ok(next);for(const field of ['left','top','width','height'])assert.ok(Math.abs(old.rect[field]-next.rect[field])<=.05,profile+' '+old.path+' '+field);}
 console.log(JSON.stringify({profile,judgmentBlocksCompared:blocks.length,normalGeometry:'preserved'}));
}
