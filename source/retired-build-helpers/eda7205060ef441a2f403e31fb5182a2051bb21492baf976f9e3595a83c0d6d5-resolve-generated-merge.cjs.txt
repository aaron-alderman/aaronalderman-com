const cp=require('child_process');const git=(...args)=>cp.execFileSync('git',['-c','safe.directory='+process.cwd().replaceAll('\\','/'),...args],{encoding:'utf8'});const conflicts=git('diff','--name-only','--diff-filter=U').trim().split(/\r?\n/).filter(Boolean);if(conflicts.some(f=>!f.startsWith('dist/')))throw Error('Source merge requires review: '+conflicts);for(const file of conflicts)git('checkout','--ours','--',file);console.log('Regenerating '+conflicts.length+' output conflicts from the merged source.');
