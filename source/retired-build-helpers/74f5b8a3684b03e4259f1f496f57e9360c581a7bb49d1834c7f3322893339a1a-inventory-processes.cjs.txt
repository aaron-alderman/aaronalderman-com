const fs=require('fs');const profile=JSON.parse(fs.readFileSync('source/process-authoring/profile.json')),manifest=JSON.parse(fs.readFileSync('bootstrap/site.json')),supported=['let','when','port','fold','assert','call','try'];
const rows=[];for(const name of fs.readdirSync('algebra').filter(f=>f.endsWith('.process.json'))){const file='algebra/'+name,module=JSON.parse(fs.readFileSync(file)),forms=new Map(),operators=new Set(),ports=new Set(),issues=[];let statements=0;
 function expression(v){if(Array.isArray(v)){operators.add(v[0]);v.slice(1).forEach(expression);}else if(v&&typeof v==='object')Object.values(v).forEach(expression);}
 function programme(body){for(const st of body){statements++;const kinds=['let','when','port','fold','assert','call','try','each'].filter(k=>Object.hasOwn(st,k));for(const k of kinds)forms.set(k,(forms.get(k)??0)+1);if(kinds.length!==1)issues.push('unknown statement');if(st.port)ports.add(st.port);for(const key of ['value','when','args','fold','initial','yield','assert','each'])if(Object.hasOwn(st,key))expression(st[key]);if(st.assert&&typeof st.message!=='string')issues.push('computed assertion message');if(st.call&&typeof st.call!=='string')issues.push('computed direct call');for(const k of ['then','steps','try','catch'])if(st[k])programme(st[k]);}}
 Object.values(module.queries).forEach(programme);rows.push({file,active:manifest.modules.includes(file),queries:Object.keys(module.queries).length,statements,bytes:fs.statSync(file).size,forms:Object.fromEntries(forms),missingForms:[...forms.keys()].filter(x=>!supported.includes(x)),missingOperators:[...operators].filter(x=>!profile.operators.includes(x)),additionalPorts:[...ports].filter(x=>!profile.directPorts.includes(x)),issues:[...new Set(issues)]});}
fs.writeFileSync('.build/process-inventory.json',JSON.stringify(rows,null,2));console.log(JSON.stringify({remaining:rows.length,active:rows.filter(r=>r.active).length,queryCount:rows.reduce((a,r)=>a+r.queries,0),syntaxReady:rows.filter(r=>!r.missingForms.length&&!r.missingOperators.length&&!r.issues.length).map(r=>r.file),each:rows.filter(r=>r.missingForms.includes('each')).map(r=>r.file),operatorGaps:[...new Set(rows.flatMap(r=>r.missingOperators))],otherIssues:rows.filter(r=>r.issues.length).map(r=>[r.file,r.issues])},null,2));
