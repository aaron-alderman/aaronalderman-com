const fs=require('fs');
// A periodic three-strand plait: alternating adjacent crossings, returning each
// strand to its starting lane after six crossings. The ground is fully opaque.
function tile(accent,vertical=false){const colours=['#e8bd70','#fff0bc',accent],ys=[2,6,10],order=[0,1,2],segments=[[],[],[]],overs=[];for(let s=0;s<6;s++){const lane=s%2,x=s*12,start=[...order],end=[...order];[end[lane],end[lane+1]]=[end[lane+1],end[lane]];const over=start[lane];for(let strand=0;strand<3;strand++){const a=ys[start.indexOf(strand)],b=ys[end.indexOf(strand)],d=`M${x} ${a} C${x+4} ${a} ${x+8} ${b} ${x+12} ${b}`;segments[strand].push(d);if(strand===over)overs.push({strand,d});}order.splice(0,3,...end);}
const path=(d,c)=>`<path d="${d}" fill="none" stroke="#785332" stroke-width="5.8"/><path d="${d}" fill="none" stroke="${c}" stroke-width="4.8"/><path d="${d}" fill="none" stroke="#fff6d9" stroke-width=".65" opacity=".65"/>`;
const content='<rect width="72" height="12" fill="#b28b50"/>'+segments.map((ss,i)=>path(ss.join(' '),colours[i])).join('')+overs.map(x=>path(x.d,colours[x.strand])).join('');return `<svg xmlns="http://www.w3.org/2000/svg" width="${vertical?12:72}" height="${vertical?72:12}" viewBox="0 0 ${vertical?'12 72':'72 12'}"><g stroke-linecap="butt" stroke-linejoin="round"${vertical?' transform="translate(12 0) rotate(90)"':''}>${content}</g></svg>`;}
const url=svg=>'url("data:image/svg+xml,'+encodeURIComponent(svg).replaceAll("'",'%27')+'")';let rel=fs.readFileSync('source/hierarchy.rel','utf8'),i=0;const put=(s,p,o)=>{rel+='('+[s,p,o].map(JSON.stringify).join(', ')+')\n';};function style(selector,props){const s='triple-braid.style.'+(++i);put(s,'isa','StyleRule');put(s,'projection','main');put(s,'selector',selector);for(const[k,v]of Object.entries(props)){put(s,'declaration',s+'.'+k);put(s+'.'+k,'property',k);put(s+'.'+k,'value',v);}}
for(const[id,c]of [['first-trails','#9fc8e4'],['threads','#e4a9c2'],['atlas','#bdb1ee']])style('.begin-door#'+id,{'--braid-horizontal':url(tile(c)),'--braid-vertical':url(tile(c,true))});
style('.begin-door::before',{padding:'10px',background:'var(--braid-horizontal),var(--braid-horizontal),var(--braid-vertical),var(--braid-vertical),linear-gradient(#bd9558,#bd9558)','background-size':'72px 10px,72px 10px,10px 72px,10px 72px,100% 100%','background-position':'0 0,0 100%,0 0,100% 0,0 0','background-repeat':'repeat-x,repeat-x,repeat-y,repeat-y,no-repeat',opacity:'1'});
style('.begin-door::after',{inset:'10px','border-radius':'5px',border:'1px solid #edcc8c77'});
style('.begin-door:hover::before,.begin-door:focus-visible::before',{'background-position':'12px 0,-12px 100%,0 12px,100% -12px,0 0',filter:'brightness(1.1)'});
fs.writeFileSync('source/hierarchy.rel',rel);for(const file of ['algebra/hierarchy.process.json','algebra-publication.test.mjs'])fs.writeFileSync(file,fs.readFileSync(file,'utf8').replaceAll('expedition.css?v=braided-1','expedition.css?v=triple-braid-1'));const p='source/expedition.rel';fs.writeFileSync(p,fs.readFileSync(p,'utf8').replace('"stylesheetVersion", "braided-1"','"stylesheetVersion", "triple-braid-1"'));
